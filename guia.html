1) Crear el proyecto Web Forms

  1. Visual Studio 2022 → Create a new project → ASP.NET Web Application (.NET Framework).

  2. Nombre: Escuela.Web. Framework: .NET Framework 4.8.

  3. Plantilla: Web Forms (puedes desmarcar autenticación si no la necesitas).

2) Instalar EF6 (Entity Framework 6)

Menú Tools → NuGet Package Manager → Package Manager Console.
Asegúrate que Default project = Escuela.Web y ejecuta:

Install-Package EntityFramework

Verifica en References: EntityFramework y EntityFramework.SqlServer.
En Web.config debe aparecer una sección <entityFramework> (si no, la agregamos abajo).

3.1 Models/Estudiante.cs

using System;
using System.ComponentModel.DataAnnotations;

namespace Escuela.Web.Models
{
    public class Estudiante
    {
        [Key]
        public int IdEstudiante { get; set; }

        [Required, StringLength(100)]
        public string Nombre { get; set; }

        [StringLength(200), EmailAddress]
        public string Email { get; set; }

        [Required(ErrorMessage = "La edad es obligatoria")]
        [Range(18, 40, ErrorMessage = "La edad debe estar entre 18 y 40")]
        public int Edad { get; set; }

        public DateTime Creado { get; set; }
    }
}

3.2 Data/EscuelaContext.cs

using System.Data.Entity;
using System.Data.Entity.ModelConfiguration.Conventions;
using Escuela.Web.Models;

namespace Escuela.Web.Data
{
    public class EscuelaContext : DbContext
    {
        // Debe existir en Web.config: name=EscuelaConnection
        public EscuelaContext() : base("name=EscuelaConnection") { }

        public DbSet<Estudiante> Estudiantes { get; set; }

        protected override void OnModelCreating(DbModelBuilder modelBuilder)
        {
            modelBuilder.Conventions.Remove<PluralizingTableNameConvention>();

            // Mapea a la tabla existente en el esquema 'escolar'
            modelBuilder.Entity<Estudiante>()
                        .ToTable("estudiantes", schemaName: "escolar");

            // No permitir que EF altere el esquema existente
            Database.SetInitializer<EscuelaContext>(null);

            base.OnModelCreating(modelBuilder);
        }
    }
}

3.3 Services/EstudianteService.cs

using System;
using System.Collections.Generic;
using System.Linq;
using Escuela.Web.Data;
using Escuela.Web.Models;

namespace Escuela.Web.Services
{
    /// <summary>
    /// Capa de negocio para Estudiante. Las páginas deben usar este servicio,
    /// no el DbContext directamente.
    /// </summary>
    public class EstudianteService
    {
        // ---------- Consultas ----------

        public IEnumerable<Estudiante> Listar(string filtro = null)
        {
            using (var ctx = new EscuelaContext())
            {
                var q = ctx.Estudiantes.AsQueryable();

                if (!string.IsNullOrWhiteSpace(filtro))
                {
                    filtro = filtro.Trim();
                    q = q.Where(e =>
                        e.Nombre.Contains(filtro) ||
                        (e.Email != null && e.Email.Contains(filtro)));
                }

                return q.OrderBy(e => e.IdEstudiante).ToList();
            }
        }

        public Estudiante Obtener(int id)
        {
            using (var ctx = new EscuelaContext())
            {
                return ctx.Estudiantes.Find(id);
            }
        }

        // ---------- Comandos (CRUD) ----------

        public void Crear(Estudiante e)
        {
            if (e == null) throw new ArgumentNullException(nameof(e));
            NormalizarYValidar(e);

            using (var ctx = new EscuelaContext())
            {
                ValidarUnicidadEmail(ctx, e.Email);
                ctx.Estudiantes.Add(e);
                ctx.SaveChanges(); // 'Creado' lo asigna la BD por DEFAULT SYSUTCDATETIME()
            }
        }

        public void Actualizar(Estudiante cambios)
        {
            if (cambios == null) throw new ArgumentNullException(nameof(cambios));
            NormalizarYValidar(cambios);

            using (var ctx = new EscuelaContext())
            {
                var existente = ctx.Estudiantes.Find(cambios.IdEstudiante);
                if (existente == null)
                    throw new KeyNotFoundException("El estudiante no existe.");

                // Si cambia email, valida unicidad
                if (!string.Equals(existente.Email ?? "", cambios.Email ?? "", StringComparison.Ordinal))
                {
                    ValidarUnicidadEmail(ctx, cambios.Email, excluirId: cambios.IdEstudiante);
                }

                existente.Nombre = cambios.Nombre;
                existente.Email  = cambios.Email;   // puede ser null
                existente.Edad   = cambios.Edad;

                ctx.SaveChanges();
            }
        }

        public void Eliminar(int id)
        {
            using (var ctx = new EscuelaContext())
            {
                var e = ctx.Estudiantes.Find(id);
                if (e == null) return; // idempotente

                ctx.Estudiantes.Remove(e);
                ctx.SaveChanges();
            }
        }

        // ---------- Reglas de negocio ----------

        private static void NormalizarYValidar(Estudiante e)
        {
            e.Nombre = (e.Nombre ?? "").Trim();
            if (string.IsNullOrWhiteSpace(e.Nombre))
                throw new InvalidOperationException("El nombre es obligatorio.");

            e.Email = string.IsNullOrWhiteSpace(e.Email) ? null : e.Email.Trim();

            if (e.Edad < 18 || e.Edad > 40)
                throw new InvalidOperationException("La edad debe estar entre 18 y 40.");
        }

        private static void ValidarUnicidadEmail(EscuelaContext ctx, string email, int? excluirId = null)
        {
            if (string.IsNullOrEmpty(email)) return;

            var existe = ctx.Estudiantes.Any(x => x.Email == email &&
                                                 (!excluirId.HasValue || x.IdEstudiante != excluirId.Value));
            if (existe)
                throw new InvalidOperationException("El email ya está registrado.");
        }
    }
}

4) Web.config: cadena de conexión y provider EF

Abre Web.config y agrega/ajusta dentro de <configuration>:

<connectionStrings>
  <add name="EscuelaConnection"
       connectionString="Data Source=localhost\SQLEXPRESS;
                         Initial Catalog=Escuela;
                         Integrated Security=True;
                         MultipleActiveResultSets=True"
       providerName="System.Data.SqlClient" />
</connectionStrings>

<entityFramework>
  <providers>
    <provider invariantName="System.Data.SqlClient"
              type="System.Data.Entity.SqlServer.SqlProviderServices, EntityFramework.SqlServer" />
  </providers>
</entityFramework>

Si usarás login SQL:

Data Source=localhost\SQLEXPRESS;Initial Catalog=Escuela;User ID=appUser;Password=
Clave_123!;MultipleActiveResultSets=True

Students.aspx.cs

using System;
using System.Web.UI.WebControls;
using Escuela.Web.Services;

namespace Escuela.Web
{
    public partial class Students : System.Web.UI.Page
    {
        private readonly EstudianteService _service = new EstudianteService();

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                BindGrid();
            }
        }

        private void BindGrid(string filtro = null)
        {
            gvEstudiantes.DataSource = _service.Listar(filtro);
            gvEstudiantes.DataBind();
        }

        protected void btnBuscar_Click(object sender, EventArgs e)
        {
            BindGrid(txtFiltro.Text);
        }

        protected void btnLimpiar_Click(object sender, EventArgs e)
        {
            txtFiltro.Text = string.Empty;
            BindGrid();
        }

        protected void gvEstudiantes_PageIndexChanging(object sender, GridViewPageEventArgs e)
        {
            // Aseguramos que el índice de página esté dentro del rango
            if (e.NewPageIndex >= 0 && e.NewPageIndex < gvEstudiantes.PageCount)
            {
                gvEstudiantes.PageIndex = e.NewPageIndex;
                BindGrid(txtFiltro.Text);
            }
        }

        protected void gvEstudiantes_RowEditing(object sender, GridViewEditEventArgs e)
        {
            // Validar índice antes de usarlo
            if (e.NewEditIndex >= 0 && e.NewEditIndex < gvEstudiantes.DataKeys.Count)
            {
                int id = (int)gvEstudiantes.DataKeys[e.NewEditIndex].Value;
                Response.Redirect($"~/StudentForm.aspx?id={id}");
            }
        }

        protected void gvEstudiantes_RowDeleting(object sender, GridViewDeleteEventArgs e)
        {
            // Validar índice antes de usarlo
            if (e.RowIndex >= 0 && e.RowIndex < gvEstudiantes.DataKeys.Count)
            {
                int id = (int)gvEstudiantes.DataKeys[e.RowIndex].Value;
                try
                {
                    _service.Eliminar(id);
                    BindGrid(txtFiltro.Text);
                    lblMsg.Text = string.Empty;
                }
                catch (Exception ex)
                {
                    lblMsg.Text = "No se pudo eliminar el registro: " + ex.Message;
                }
            }
        }

        protected void gvEstudiantes_RowDataBound(object sender, GridViewRowEventArgs e)
        {
            if (e.Row.RowType == DataControlRowType.DataRow)
            {
                // Buscar de forma segura el LinkButton de Delete en TODA la fila
                foreach (Control control in e.Row.Cells[e.Row.Cells.Count - 1].Controls)
                {
                    if (control is LinkButton lb && lb.CommandName == "Delete")
                    {
                        lb.OnClientClick = "return confirm('¿Eliminar este estudiante?');";
                    }
                }
            }
        }
    }
}

StudentForm.aspx.cs

using System;
using Escuela.Web.Models;
using Escuela.Web.Services;

namespace Escuela.Web
{
    public partial class StudentForm : System.Web.UI.Page
    {
        private readonly EstudianteService _service = new EstudianteService();
        private int? Id => int.TryParse(Request.QueryString["id"], out var x) ? x : (int?)null;

        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                litTitulo.Text = Id.HasValue ? "Editar Estudiante" : "Nuevo Estudiante";
                if (Id.HasValue) Cargar();
            }
        }

        private void Cargar()
        {
            var e = _service.Obtener(Id.Value);
            if (e == null)
            {
                Response.Redirect("~/Students.aspx");
                return;
            }

            txtNombre.Text = e.Nombre;
            txtEmail.Text  = e.Email;
            txtEdad.Text   = e.Edad.ToString();
        }

        protected void btnGuardar_Click(object sender, EventArgs e)
        {
            if (!Page.IsValid) return;

            try
            {
                if (!int.TryParse(txtEdad.Text, out var edad))
                {
                    lblMsg.Text = "La edad debe ser un número válido.";
                    return;
                }

                if (Id.HasValue)
                {
                    var cambios = new Estudiante
                    {
                        IdEstudiante = Id.Value,
                        Nombre = txtNombre.Text,
                        Email  = string.IsNullOrWhiteSpace(txtEmail.Text) ? null : txtEmail.Text,
                        Edad   = edad
                    };
                    _service.Actualizar(cambios);
                }
                else
                {
                    var nuevo = new Estudiante
                    {
                        Nombre = txtNombre.Text,
                        Email  = string.IsNullOrWhiteSpace(txtEmail.Text) ? null : txtEmail.Text,
                        Edad   = edad
                        // 'Creado' lo asigna la BD por DEFAULT
                    };
                    _service.Crear(nuevo);
                }

                Response.Redirect("~/Students.aspx");
            }
            catch (InvalidOperationException ex)
            {
                // Reglas de negocio (nombre vacío, edad fuera de rango, email duplicado)
                lblMsg.Text = ex.Message;
            }
            catch (Exception ex)
            {
                lblMsg.Text = "Error al guardar: " + ex.Message;
            }
        }
    }
}

7) Checklist / Errores comunes

  • EscuelaContext no se encuentra → faltan using Escuela.Web.Data; o el namespace no coincide.

  • “Invalid object name escolar.estudiantes” → la tabla no existe o está en otro esquema; confirma ToTable("estudiantes","escolar").

  • Login failed → revisa cadena de conexión (servidor/credenciales).

  • Email duplicado → el Service lanza InvalidOperationException y se muestra en lblMsg.
